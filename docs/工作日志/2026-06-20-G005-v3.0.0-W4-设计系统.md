## 六、应用模式(给后续 89 页面用)

```typescript
// 1. 引入
import {
  AppLayout,
  ProTable,
  AppStatistic,
  AppEmpty,
  AppToast,
  PageContainer,
  CardSection,
} from '@components/antd';

// 2. 页面
function MyPage() {
  const toast = useToast();
  const { confirm } = useConfirm();

  return (
    <PageContainer title="标题" extra={<Button>操作</Button>}>
      <CardSection title="数据">
        <ProTable dataSource={data} columns={columns} rowKey="id" />
      </CardSection>
    </PageContainer>
  );
}

// 3. 在路由中包裹
<Route path="/xxx" element={
  <AppLayout sidebarItems={items} user={user} onLogout={...}>
    <MyPage />
  </AppLayout>
} />
```
