// src/machines/reportMachine.ts
import { createMachine } from 'xstate';

export const reportMachine = createMachine({
  id: 'report',
  initial: 'pendingAssignment',
  states: {
    pendingAssignment: { on: { ASSIGN: 'assigned', WITHDRAW: 'withdrawn' } },
    assigned: { on: { START: 'writing', WITHDRAW: 'withdrawn' } },
    writing: { on: { SAVE: 'writing', SUBMIT: 'submitted' } },
    submitted: { on: { REVIEW_START: 'reviewing', REJECT: 'rejected' } },
    reviewing: { on: { APPROVE: 'reviewed', REJECT: 'rejected' } },
    reviewed: { on: { SIGN_START: 'signing' } },
    signing: { on: { SIGN_COMPLETE: 'signed' } },
    signed: { on: { PUBLISH: 'published' } },
    published: { on: { AMEND: 'amending', ARCHIVE: 'archived' } },
    amending: { on: { AMEND_COMPLETE: 'amended' } },
    amended: { on: { ARCHIVE: 'archived' } },
    rejected: { on: { RESTART: 'writing' } },
    withdrawn: { type: 'final' },
    archived: { type: 'final' },
  },
